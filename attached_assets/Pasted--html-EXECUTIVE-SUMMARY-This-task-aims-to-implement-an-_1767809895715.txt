// src/components/AlertBar.tsx
import React, { useState, useEffect } from 'react';
import { getAlert } from '../services/alertService';

interface Alert {
    id: string;
    message: string;
    isActive: boolean;
}

const AlertBar: React.FC = () =&gt; {
    const [alert, setAlert] = useState(null);
    const [isDismissed, setIsDismissed] = useState(localStorage.getItem('alertDismissed') === 'true');

    useEffect(() =&gt; {
        const fetchAlert = async () =&gt; {
            try {
                const data = await getAlert();
                setAlert(data);
            } catch (error) {
                console.error("Error fetching alert:", error);
            }
        };

        fetchAlert();
    }, []);

    const handleDismiss = () =&gt; {
        setIsDismissed(true);
        localStorage.setItem('alertDismissed', 'true');
    };

    if (!alert || !alert.message || isDismissed && !alert.isActive) {
        return null;
    }

    const backgroundColor = alert.isActive ? 'red' : 'green';

    return (
        &lt;div style={{ backgroundColor, color: 'white', padding: '10px', textAlign: 'center', position: 'fixed', top: 0, left: 0, width: '100%', zIndex: 1000 }}&gt;
            {alert.message}
            {!alert.isActive && (
                &lt;button onClick={handleDismiss} style={{ marginLeft: '10px', cursor: 'pointer' }}&gt;
                    X
                &lt;/button&gt;
            )}
        &lt;/div&gt;
    );
};

export default AlertBar;

// src/admin/AdminAlert.tsx
import React, { useState, useEffect } from 'react';
import { createAlert, updateAlert, getAlert } from '../services/alertService';

const AdminAlert: React.FC = () =&gt; {
    const [message, setMessage] = useState('');
    const [isActive, setIsActive] = useState(false);
    const [alertId, setAlertId] = useState('');

    useEffect(() =&gt; {
        const fetchAlert = async () =&gt; {
            try {
                const alert = await getAlert();
                if (alert) {
                    setMessage(alert.message);
                    setIsActive(alert.isActive);
                    setAlertId(alert.id);
                }
            } catch (error) {
                console.error("Error fetching alert:", error);
            }
        };

        fetchAlert();
    }, []);

    const handleSubmit = async (e: React.FormEvent) =&gt; {
        e.preventDefault();

        try {
            if (alertId) {
                await updateAlert(alertId, message, isActive);
            } else {
                await createAlert(message, isActive);
            }
            alert("Alert updated successfully!");
        } catch (error) {
            console.error("Error updating alert:", error);
            alert("Error updating alert. Please check the console.");
        }
    };

    return (
        &lt;form onSubmit={handleSubmit}&gt;
            &lt;label&gt;
                Message:
                &lt;input
                    type="text"
                    value={message}
                    onChange={(e) =&gt; setMessage(e.target.value)}
                    maxLength={255}
                /&gt;
            &lt;/label&gt;
            &lt;label&gt;
                Active:
                &lt;input
                    type="checkbox"
                    checked={isActive}
                    onChange={(e) =&gt; setIsActive(e.target.checked)}
                /&gt;
            &lt;/label&gt;
            &lt;button type="submit"&gt;Save Alert&lt;/button&gt;
        &lt;/form&gt;
    );
};

export default AdminAlert;
