-- Part 2: Raw math vs. what the VIEW reports for the active lots
SELECT
    lb.lot_id,
    lb.lot_number,
    lb.sku_code,
    lb.balance                                                             AS view_balance,
    SUM(it.quantity) FILTER (WHERE it.transaction_type IN ('Receive','Adjust Up','Repack'))  AS total_added,
    SUM(it.quantity) FILTER (WHERE it.transaction_type IN ('Ship','Adjust Down'))            AS total_deducted,
    SUM(it.quantity) FILTER (WHERE it.transaction_type IN ('Receive','Adjust Up','Repack'))
      - COALESCE(SUM(it.quantity) FILTER (WHERE it.transaction_type IN ('Ship','Adjust Down')), 0)
                                                                           AS calculated_balance
FROM lot_balances lb
LEFT JOIN inventory_transactions it ON it.lot_id = lb.lot_id
WHERE lb.lot_id IN (30, 11, 29, 7)
GROUP BY lb.lot_id, lb.lot_number, lb.sku_code, lb.balance
ORDER BY lb.lot_id;
